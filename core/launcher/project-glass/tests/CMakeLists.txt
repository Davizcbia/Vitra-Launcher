cmake_minimum_required(VERSION 3.25)
project(ProjectGlassEditorTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
find_package(Qt6 REQUIRED COMPONENTS Core Gui Qml Quick QuickTest QuickControls2 ShaderTools)
set(NBT_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(NBT_USE_ZLIB OFF CACHE BOOL "" FORCE)
add_subdirectory(../../../libraries/libnbtplusplus nbt)
add_executable(InstanceEditorFilesTest InstanceEditorFilesTest.cpp)
target_link_libraries(InstanceEditorFilesTest PRIVATE Qt6::Core nbt++)
enable_testing()
add_test(NAME InstanceEditorFilesTest COMMAND InstanceEditorFilesTest)
add_executable(LaunchDiagnosisTest LaunchDiagnosisTest.cpp)
target_link_libraries(LaunchDiagnosisTest PRIVATE Qt6::Core)
add_test(NAME LaunchDiagnosisTest COMMAND LaunchDiagnosisTest)
add_executable(SyncFilesTest SyncFilesTest.cpp)
target_link_libraries(SyncFilesTest PRIVATE Qt6::Core)
add_test(NAME SyncFilesTest COMMAND SyncFilesTest)
add_executable(ModpackDropTest ModpackDropTest.cpp)
target_link_libraries(ModpackDropTest PRIVATE Qt6::Core)
add_test(NAME ModpackDropTest COMMAND ModpackDropTest)
add_executable(ThemeStoreTest ThemeStoreTest.cpp ../ThemeStore.cpp ../ThemeStore.h)
set_target_properties(ThemeStoreTest PROPERTIES AUTOMOC ON)
target_link_libraries(ThemeStoreTest PRIVATE Qt6::Core Qt6::Gui)
add_test(NAME ThemeStoreTest COMMAND ThemeStoreTest)
add_executable(ThemeUiTest ThemeUiTest.cpp ../ThemeStore.cpp ../ThemeStore.h ../MacWindowMetrics.h ../../resources/backgrounds/backgrounds.qrc)
set_target_properties(ThemeUiTest PROPERTIES AUTOMOC ON AUTORCC ON)
target_link_libraries(ThemeUiTest PRIVATE Qt6::QuickTest Qt6::Quick Qt6::Qml Qt6::QuickControls2)
# The drag integration test uses the Qt platform event entry point so it
# exercises window routing rather than calling the DropArea handler directly.
get_target_property(_qt_gui_headers Qt6::Gui INTERFACE_INCLUDE_DIRECTORIES)
get_target_property(_qt_core_headers Qt6::Core INTERFACE_INCLUDE_DIRECTORIES)
foreach(_header_dir IN LISTS _qt_gui_headers _qt_core_headers)
    if(EXISTS "${_header_dir}/${Qt6_VERSION}")
        get_filename_component(_qt_module_name "${_header_dir}" NAME)
        target_include_directories(ThemeUiTest PRIVATE "${_header_dir}/${Qt6_VERSION}" "${_header_dir}/${Qt6_VERSION}/${_qt_module_name}")
        if(EXISTS "${_header_dir}/${Qt6_VERSION}/QtGui/qpa/qwindowsysteminterface.h")
            target_compile_definitions(ThemeUiTest PRIVATE VITRA_HAS_WINDOW_DRAG_TEST=1)
        endif()
    endif()
endforeach()
target_compile_definitions(ThemeUiTest PRIVATE FRONTEND_SOURCE_DIR="${CMAKE_CURRENT_SOURCE_DIR}/.." NOMINMAX)
qt_add_shaders(ThemeUiTest review_shaders PREFIX "/pgshaders" BASE "../shaders"
    FILES ../shaders/continuousclip.frag ../shaders/sidebarglow.frag ../shaders/liquidglass.frag)
qt_add_resources(ThemeUiTest review_art PREFIX "/qt/qml/ProjectGlass" BASE ".."
    FILES ../assets/default-account.svg ../assets/default-fabric.svg ../assets/default-neoforge.svg
          ../assets/default-quilt.svg ../assets/default-forge.png ../assets/default-vanilla.png ../assets/version-grass.png)
add_test(NAME ThemeUiTest COMMAND ThemeUiTest -input ${CMAKE_CURRENT_SOURCE_DIR}/themes-ui -o -,txt)
set_tests_properties(ThemeUiTest PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=windows;QSG_RHI_BACKEND=d3d11;QT_WIN_DEBUG_CONSOLE=attach")
